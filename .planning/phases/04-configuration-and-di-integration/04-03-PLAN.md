    ```go
    // NewPromptManagerProvider creates a PromptManager via DI.

    }
    ```

    ```go
    func TestNewPromptManagerProvider(t *testing.T) {
        // Test with memory store (default)
        t.Run("memory store", func(t *testing.T) {
            injector := do.New()
            do.Provide(injector, config.NewService)
            do.Provide(injector, logger.NewNopLogger)
            do.MustInvoke[*config.ServiceImpl](injector)

            // Register store provider
            do.Provide(injector, store.NewPromptStore)

            // Create manager via provider
            pm, err := manager.NewPromptManagerProvider(injector)
            require.NoError(t, err)
            assert.NotNil(t, pm)

            // Verify manager works
            ctx := context.Background()
            prompts, err := pm.ListPrompts(ctx, store.ListFilter{})
            require.NoError(t, err)
            assert.NotNil(t, prompts)
        })

        // Test with file store
        t.Run("file store", func(t *testing.T) {
            t.Setenv("GOLLUM_PROMPT_STORE_TYPE", "file")
            t.Setenv("GOLLUM_PROMPT_STORE_FILE_PATH", t.TempDir())

            injector := do.New()
            do.Provide(injector, config.NewService)
            do.Provide(injector, logger.NewNopLogger)
            do.MustInvoke[*config.ServiceImpl](injector)

            do.Provide(injector, store.NewPromptStore)

            pm, err := manager.NewPromptManagerProvider(injector)
            require.NoError(t, err)
            assert.NotNil(t, pm)
        })
    }
    ```
