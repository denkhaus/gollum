    Add PromptOptimizerConfig struct to pkg/config/service.go:

    ```go
    // OptimizerStrategy defines the optimization approach.
    // Matches optimizer.OptimizerStrategy but in config package to avoid import cycle.
    type OptimizerStrategy string

    const (
        OptimizerStrategyGradient     OptimizerStrategy = "gradient"
        OptimizerStrategyMetaPrompt   OptimizerStrategy = "metaprompt"
        OptimizerStrategyPromptMemory OptimizerStrategy = "prompt_memory"
    )

    // PromptOptimizerConfig contains configuration for prompt optimizer.
    type PromptOptimizerConfig struct {
        // DefaultStrategy is the optimization strategy to use (gradient, metaprompt, prompt_memory)
        DefaultStrategy OptimizerStrategy `envconfig:"DEFAULT_STRATEGY" default:"gradient"`

        // DefaultProvider is the LLM provider to use for optimization (anthropic, openai, gemini)
        DefaultProvider string `envconfig:"DEFAULT_PROVIDER" default:"anthropic"`

        // MaxReflectionSteps is the maximum number of reflection iterations (default: 5)
        MaxReflectionSteps int `envconfig:"MAX_REFLECTION_STEPS" default:"5"`

        // MinReflectionSteps is the minimum number of reflection iterations (default: 1)
        MinReflectionSteps int `envconfig:"MIN_REFLECTION_STEPS" default:"1"`
    }
    ```

    Add PromptOptimizerConfig field to serviceImpl struct:
    ```go
    type serviceImpl struct {
        // ... existing fields ...
        PromptStore PromptStoreConfig `envconfig:"PROMPT_STORE"`
        Optimizer   PromptOptimizerConfig `envconfig:"OPTIMIZER"`
    }
    ```

    Add GetPromptOptimizerConfig to ConfigService interface and implementation:

        // ... existing methods ...
        GetPromptOptimizerConfig() *PromptOptimizerConfig

        return &s.Optimizer
    }
