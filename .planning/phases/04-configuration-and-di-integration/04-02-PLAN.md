    ```go
    package optimizer

    import (
        "context"
        "fmt"

        "github.com/denkhaus/gollum/pkg/config"
        "github.com/denkhaus/gollum/pkg/llm"
        "github.com/denkhaus/gollum/pkg/prompt/manager"
        "github.com/denkhaus/gollum/pkg/shared"
        "github.com/samber/do/v2"
    )

    // NewOptimizerProvider creates a PromptOptimizer via DI.
    // Uses config from ConfigService and LLM client from ClientProvider.
    func NewOptimizerProvider(injector do.Injector) (PromptOptimizer, error) {
        // Get configuration service
        configService := do.MustInvoke[config.ConfigService](injector)

        // Get LLM client provider
        clientProvider := do.MustInvoke[llm.ClientProvider](injector)

        // Get prompt manager
        promptManager := do.MustInvoke[manager.PromptManager](injector)

        // Load optimizer config
        cfg := configService.GetPromptOptimizerConfig()

        // Map config provider string to shared.LLMProvider
        var provider shared.LLMProvider
        switch cfg.DefaultProvider {
        case "anthropic":
            provider = shared.LLMProviderAnthropic
        case "openai":
            provider = shared.LLMProviderOpenAI
        case "gemini":
            provider = shared.LLMProviderGemini
        default:
            return nil, fmt.Errorf("unknown LLM provider: %s", cfg.DefaultProvider)
        }

        // Get LLM client
        ctx := context.Background()
        client, err := clientProvider.GetClient(ctx, provider)
        if err != nil {
            return nil, fmt.Errorf("failed to get LLM client: %w", err)
        }

        // Map config strategy to optimizer strategy
        var strategy OptimizerStrategy
        switch cfg.DefaultStrategy {
        case "gradient":
            strategy = StrategyGradient
        case "metaprompt":
            strategy = StrategyMetaPrompt
        case "prompt_memory":
            strategy = StrategyPromptMemory
        default:
            return nil, fmt.Errorf("unknown optimizer strategy: %s", cfg.DefaultStrategy)
        }

        // Create optimizer config
        optimizerCfg := &OptimizerConfig{
            Kind:               strategy,
            Provider:           cfg.DefaultProvider,
            MaxReflectionSteps: cfg.MaxReflectionSteps,
            MinReflectionSteps: cfg.MinReflectionSteps,
        }

        // Create and return optimizer
        return NewOptimizer(client, promptManager, optimizerCfg)
    }
    ```

    IMPORTANT patterns from guide.golang.di.md:
    - Constructor signature: func NewXXXProvider(injector do.Injector) (XXX, error)
    - Use do.MustInvoke to get dependencies from DI container
    - Return interface type (PromptOptimizer), not implementation
    - Handle errors from config mapping and client creation
