    ```go
    // getTraceContext retrieves the TraceContext for a given session ID.
    // Returns nil if no trace context exists for this session.
    // Uses RLock to allow concurrent reads.
    func (h *LangfuseHook) getTraceContext(sessionID uuid.UUID) *TraceContext {
        h.traceCtxsMu.RLock()
        defer h.traceCtxsMu.RUnlock()

        return h.traceCtxs[sessionID]
    }
    ```

    This method provides read-only access to trace contexts with proper locking. The nil return when not found is consistent with Go idioms and allows callers to distinguish "no context" from "empty context".

    ```go
    // createTraceContext creates a new TraceContext for the given session.
    // Generates a unique trace ID and initializes the span map.
    // Uses Lock for exclusive write access.
    func (h *LangfuseHook) createTraceContext(sessionID uuid.UUID) *TraceContext {
        h.traceCtxsMu.Lock()
        defer h.traceCtxsMu.Unlock()

        // Generate unique trace ID using Langfuse SDK
        client, err := h.getClient()
        if err != nil {
            // If client not available, use UUID as fallback
            traceID := uuid.New()
            tc := &TraceContext{
                TraceID:   traceID.String(),
                RootSpan:  nil,
                Spans:     make(map[string]interface{}),
                SessionID: sessionID,
                CreatedAt:  time.Now(),
            }
            h.traceCtxs[sessionID] = tc
            return tc
        }

        // Create Langfuse trace (will store RootSpan in Phase 8)
        // For now, create minimal context - actual trace creation in Phase 10
        traceID := uuid.New()
        tc := &TraceContext{
            TraceID:   traceID.String(),
            RootSpan:  nil, // Set in Phase 10 (Session Tracing)
            Spans:     make(map[string]interface{}),
            SessionID: sessionID,
            CreatedAt:  time.Now(),
        }
        h.traceCtxs[sessionID] = tc

        return tc
    }
    ```

    NOTE: In Phase 10 (Session Tracing), RootSpan will be populated with actual Langfuse trace creation. For now, it's nil to indicate no root span created yet.

    Error handling: If getClient() fails (no credentials), we still create a TraceContext with a UUID-based trace ID. This prevents trace hooks from failing when Langfuse is misconfigured.

    ```go
    func (h *LangfuseHook) removeTraceContext(sessionID uuid.UUID) {
        h.traceCtxsMu.Lock()
        defer h.traceCtxsMu.Unlock()

        delete(h.traceCtxs, sessionID)
    }
    ```

    This method cleans up trace context after a session ends. The delete operation on Go maps is safe even if the key doesn't exist (no-op), which simplifies session cleanup logic.

    ```go
        mockLog := mocks.NewMockLoggerService(t)
        cfg := &config.LangfuseConfig{
            LangfuseEnabled:   false,

        }

        t.Run("getTraceContext returns nil when not found", func(t *testing.T) {
            tc := hook.getTraceContext(sessionID)
            assert.Nil(t, tc, "getTraceContext should return nil for non-existent session")
        })

        t.Run("createTraceContext creates and stores context", func(t *testing.T) {
            tc := hook.createTraceContext(sessionID)

            assert.Equal(t, sessionID, tc.SessionID, "SessionID should match")
            assert.NotEmpty(t, tc.TraceID, "TraceID should be generated")
            assert.NotNil(t, tc.Spans, "Spans map should be initialized")
            assert.False(t, tc.CreatedAt.IsZero(), "CreatedAt should be set")
        })

            hook.createTraceContext(sessionID)
            tc := hook.getTraceContext(sessionID)

            assert.Equal(t, sessionID, tc.SessionID)
        })

        t.Run("removeTraceContext deletes context", func(t *testing.T) {
            hook.createTraceContext(sessionID)
            hook.removeTraceContext(sessionID)

            tc := hook.getTraceContext(sessionID)
            assert.Nil(t, tc, "getTraceContext should return nil after removal")
        })

        t.Run("concurrent access is thread-safe", func(t *testing.T) {
            // Create multiple session IDs

            session2 := uuid.New()
            session3 := uuid.New()

            go func() {

            }()
    ```
